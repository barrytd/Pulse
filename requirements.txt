# PyYAML: Reads YAML config files (pulse.yaml) for storing default settings
PyYAML>=6.0
# FastAPI: Web framework for the REST API (pulse/api.py)
fastapi>=0.100.0
# Uvicorn: ASGI server that runs the FastAPI app
uvicorn[standard]>=0.23.0

# driver automatically when they spin up a Postgres addon.

# Used by the GeoIP connector to look IPs up in a LOCAL file, so GeoIP works
